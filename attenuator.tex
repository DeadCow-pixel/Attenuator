\section{Imperfect gaussian measurements}
We begin with a brief discussion of asymmetrical coherent measurements, schematically depicted in Fig.~\ref{fig:scheme}. Asymmetry effects are induced by imbalance (unequal transmission and reflection coefficients) of the beamsplitters and unequal quantum efficiency of the detectors. For detailed derivations of the expressions in this section see Ref.~\cite{naumchik2025asymmetryeffectshomodyneheterodyne}.

$Q$ symbol of the POVM describing noisy homodyne measurements (see Fig.~\ref{subfig:homodyne}) is given by
\begin{align}
  &\label{eq:Pgood-w-def}
    {\prob}_G^{(\ind{H})}(x)=\frac{1}{\sqrt{2\pi\sigma_G}}
    \exp \left[-\frac{(x-\avr{\hat{x}_\phi})^2}{2\sigma_x}\right],\\
  &\label{eq:avr-x-phi}
  \avr{\hat{x}_\phi}\equiv \bra{\alpha}\hat{x}_\phi\ket{\alpha}={2\Re\alpha e^{-i\phi}},
  \quad \phi=\arg\alpha_L,\\
    &\label{eq:quad-op}
    \hat{x}_\phi=\hat{a}e^{-i\phi}+\hat{a}^\dag e^{i\phi}
\end{align}
where $\alpha$ ($\alpha_L$) is the complex amplitude of signal (LO) mode, $\hat{x}_\phi$ is the phase-rotated quadrature operator;
$x$ is the quadrature variable
\begin{align}
  \label{eq:x-def}
  x\equiv\frac{\mu}{(\eta_1+\eta_2)CS\left|\alpha_L\right|}-\frac{\eta_1S^2-\eta_2C^2}{\left(\eta_1+\eta_2\right)CS}\left|\alpha_L\right|
\end{align}
and $\sigma_x$ is the quadrature variance
\begin{align}
  &
        \label{eq:sigma-x}
    \sigma_x\equiv
    \frac{\eta_1S^2+\eta_2C^2}{
    \left[(\eta_1+\eta_2)CS\right]^2
    }.
\end{align}

Starting from the $Q$ symbol~\eqref{eq:Pgood-w-def} and passing through the characteristic function representation, we deduce the covariance matrix of the noisy homodyne measurements (see Ref.~\cite{naumchik2025asymmetryeffectshomodyneheterodyne}):
\begin{align}
    &\label{eq:Sigma-H-m}
    \Sigma_{m}^{(\ind{H})}= \lim_{r\to+\infty}R(\phi)\,\ind{diag}(e^{-2r}+\sigma_N,e^{2r})\,\tcnj{R}(\phi)
    %=\Sigma_{\ind{id}}^{(\ind{H})}+\Sigma_{N}^{(\ind{H})}
    ,\\
    &\label{eq:Sigma-H-id}
    \Sigma_{\ind{id}}^{(\ind{H})}=\lim_{r\to+\infty}R(\phi)\,\ind{diag}(e^{-2r},e^{2r})\,{R}^{\ind{T}}(\phi)%\\
    %&\label{eq:Sigma-N-H}
    %\Sigma_{N}^{(\ind{H})}=R(\phi)\,\ind{diag}(\sigma_N,0)\,\tcnj{R}(\phi)
    ,
\end{align}
where $\Sigma_{\ind{id}}^{(\ind{H})}$ is the covariance matrix of sharp measurement, $R(\phi)=\begin{pmatrix}\cos\phi&-\sin\phi\\ \sin\phi& \cos\phi\end{pmatrix}$ is the rotation matrix; $\sigma_N$ is the excess noise variance that accounts for asymmetry effects is given by
\begin{align}
  &
  \label{eq:sgm_N}
     \sigma_N=\sigma_x-1.
\end{align}

For double homodyne detection, $Q$ symbol is given by
\begin{align}
  \label{eq:P_G-x1x2}
  &
    \prob_G^{(\ind{DH})}(x_1,x_2)
    =\frac{1}{2\pi\sqrt{\sigma_G^{(1)}\sigma_G^{(2)}}}
    \exp\biggl[
    -\frac{(x_1-\Re\alpha e^{-i\phi})^2}{\sigma_1}
    -\frac{(x_2-\Im\alpha e^{-i\phi})^2}{\sigma_2}
    \biggr],\\
    &
    \label{eq:sigma_G-i}
    \sigma_G^{(i)}=
    (\eta_1^{(i)}S_i^2+\eta_2^{(i)}C_i^2) |\alpha_L^{(i)}|^2,\quad
    \alpha_L^{(1)}=C_L\alpha_L,\quad \alpha_L^{(2)}=-iS_L\alpha_L,
\end{align}
where $x_i$ are the quadrature variables given by
\begin{align}
  \label{eq:x-1}
  x_1&=
  \frac{1}{2(\eta_1^{(1)}+\eta_2^{(1)})C_1S_1C_S}
  \left\{
  \frac{\mu_1}{|\alpha_L^{(1)}|}-(\eta_1^{(1)}S_1^2-\eta_2^{(1)}C_1^2) |\alpha_L^{(1)}|
    \right\},
  \\
  \label{eq:x-2}
  x_2&=
  \frac{1}{2(\eta_1^{(2)}+\eta_2^{(2)})C_2S_2S_S}
  \left\{
  \frac{\mu_2}{|\alpha_L^{(2)}|}-(\eta_1^{(2)}S_2^2-\eta_2^{(2)}C_2^2) |\alpha_L^{(2)}|
    \right\},
\end{align}
and relations
\begin{align}
  &
  \label{eq:sigm-i}
    \sigma_1=\frac{\sigma_x^{(1)}}{ 2 C_S^2},
    \quad
    \sigma_2=\frac{\sigma_x^{(2)}}{ 2 S_S^2},
  \\
  &
    \label{eq:sigm-x-i}
    \sigma_x^{(i)}=\frac{\eta_1^{(i)}S_i^2+\eta_2^{(i)}C_i^2}{
    \left[(\eta_1^{(i)}+\eta_2^{(i)})C_iS_i\right]^2
    }
\end{align}
give the quadrature variances $\sigma_1$ and $\sigma_2$.

Analogously, we obtain the covariance matrix of the double homodyne POVM as:
\begin{align}
\label{eq:Sigma_m-DH}
  &
  \Sigma_m^{(\ind{DH})}=R(\phi)\diag(\delta_1,\delta_2)\tcnj{R}(\phi),
  \\
  \label{eq:sigma-i}
  &
    \delta_i=2\sigma_i-1.
\end{align}

The POVM that describes sharp double homodyne measurements uses the set of squeezed states,
\begin{align}
  \label{eq:squeezed-def}
  \ket{z,r,\phi}=\hat{R}(\phi)\hat{D}(z)\hat{S}(r)\ket{0}=
  \hat{D}(z e^{i\phi})\hat{S}(r e ^{2i\phi})\ket{0},
\end{align}
where $z$ is the coherent amplitude, $r$ is real-valued squeezing parameter, and $\phi$ is squeezing phase;
$\hat{D}(\beta)$ ($\hat{S}(\zeta)$) is the displacement (squeezing) operator  given by
\begin{align}
      \label{eq:D-def}
&\hat{D}(\beta) =
e^{\beta\hcnj{\hat{a}}-\beta^*\hat{a}},\\
\label{eq:S-def}
&\hat{S}(\zeta)=e^{\frac{1}{2}\left(\zeta\hat{a}^{\dag2}-\zeta^*\hat{a}^2\right)},
\end{align}


The covariance matrix of squeezed states reads
\begin{align}
  \label{eq:Sigma-id-DH}
  &
  \Sigma_{\ind{id}}^{(\ind{DH})}=
    R(\phi)
    \begin{pmatrix}
      e^{2r}&0\\0&e^{-2r}
    \end{pmatrix}
    \tcnj{R}(\phi),
\end{align}
and it is explicitly dependent on squeezing parameter $r$ that lies in the interval defined by measurement asymmetry (see Ref.~\cite{naumchik2025asymmetryeffectshomodyneheterodyne}):
\begin{align}
  \label{eq:r1-r2}
  &
    r\in \left[r_2,r_1\right],\;
    r_1=\ln\delta_1^{1/2},
    \;
    r_2=\ln\delta_2^{-1/2}.
\end{align}

\section{Gaussian channels modeling imperfect measurements}
Describing the measurement statistics of an arbitrary Gaussian state $\hat\rho_G$ with imperfect Gaussian measurements described by a noisy POVM $\hat\Pi_G$ amounts to computing the probability density of the outcome $x$, given by
\begin{align}
  p(x|\hat\rho_G)
  =\Tr\left[\hat\rho_G\,\hat\Pi_G^{(x)}\right]
  =\Tr\left[\mathcal{E}(\hat\rho_G)\,\hat\Pi_{\ind{id}}^{(x)}\right]
  =\Tr\left[\hat\rho_G\,\mathcal{E}^{*}(\Pi_{\ind{id}}^{(x)})\right],
\end{align}
where $\{\Pi_{\ind{id},x}\}$ is the POVM of the corresponding ideal (perfect) Gaussian measurement, and $\hat\Pi_{G}^{(x)} = \mathcal{E}^{*}(\Pi_{\ind{id}}^{(x)})$ is the effective imperfect POVM describing the noisy measurement.

An important property is that the same measurement statistics (i.e., the same probability density $p(x|\hat\rho_G)$ for all Gaussian states $\hat\rho_G$) can be described by different Gaussian channels $\mathcal{E}$, provided their dual maps $\mathcal{E}^{*}$ yield the same effective POVM $\hat\Pi_G$. Moreover, as $\hat\rho_G$, $\hat\Pi_G$, and $\mathcal{E}$ are Gaussian, everything is charactirized by the change of first and second moments.

In what follows, we will consider two distinct quantum channels to describe the imperfect measurements, namely attenuator channel and additive channel.

First, we shall use the attenuator quantum channel~\cite{Serafini:bk:2023},
characterized by matrices
\begin{align}
    \label{eq:X,Y}
    X^{(\gamma)}=\cos\theta_{\gamma}\mathbbm{1}, \quad Y^{(\gamma)}=\sin^2\theta_{\gamma}\Sigma_{\ind{aux}}^{(\gamma)}.
\end{align}

The action of the channel $\mathcal{E}_N^{(\gamma)}$ is defined as
\begin{align}
    \label{eq:E_gamma}
    \mathcal{E}_N^{(\gamma)}(\hat{\rho}) = \Tr_{\ind{aux}} \left[ \hat{U}_{\theta}^{(\gamma)} \left( \hat{\rho} \otimes \hat{\rho}_{\ind{aux}}^{(\gamma)} \right) \hat{U}_{\theta}^{(\gamma)\dagger} \right],
\end{align}
where $\hat{\rho}_{\ind{aux}}^{(\gamma)}$ is a zero-mean Gaussian state with covariance matrix $\Sigma_{\ind{aux}}^{(\gamma)}$, and $\hat{U}_{\theta}^{(\gamma)} = \exp\left[ \theta_{\gamma} (a^\dagger \hat{b} - a \hat{b}^\dagger) \right]$ is the beam-splitter unitary operator, with $\hat{a}$ and $\hat{b}$ as the annihilation operators of the system and auxiliary modes, respectively.

Dual of~\eqref{eq:E_gamma}, $\mathcal{E}_N^{*(\gamma)}$, transforms the POVM of the ideal measurement into the POVM of the noisy measurements by changing respective covariance matrix $\Sigma_{\ind{id}}^{(\gamma)}$ into $\Sigma_{m}^{(\gamma)}$ as follows:
\begin{align}
    \label{eq:psi_ast}
    \mathcal{E}_N^{*(\gamma)}:\quad \Sigma_{\ind{id}}^{(\gamma)}\mapsto\Sigma_{m}^{(\gamma)}=\frac{1}{\cos^2\theta_{\gamma}}\Sigma_{\ind{id}}^{(\gamma)}+\tan^2\theta_{\gamma}\Sigma_{\ind{aux}}^{(\gamma)},
    \quad \gamma\in\{\ind{H},\ind{DH}\}.
\end{align}

Now we derive explicit experessions coefficients of~\eqref{eq:psi_ast} (see Eq.~\eqref{eq:X,Y}).

First, we will consider homodyne detection.
It will be useful to put noisy POVM covariance matrices in the following form (see Eq.~\eqref{eq:Sigma-H-m} and~\eqref{eq:Sigma-H-id}):
\begin{align}
  &\label{eq:Sigma-H-m-dif}
    \Sigma_{m}^{(\ind{H})}=\Sigma_{\ind{id}}^{(\ind{H})}+\Sigma_{N}^{(\ind{H})}
    ,\\
    &\label{eq:Sigma-N-H}
    \Sigma_{N}^{(\ind{H})}=R(\phi)\,\ind{diag}(\sigma_N,0)\,\tcnj{R}(\phi)
    ,
\end{align}
where we explicitly separated excess noise covariance matrix $\Sigma_{N}^{(\ind{H})}$ and covariance matrix of sharp measurements $\Sigma_{\ind{id}}^{(\ind{H})}$.

Since covariance matrices of ideal and noisy measurements (see Eq.~\eqref{eq:Sigma-H-id}) are taken in limit, implying that
\begin{align}
    &\frac{1}{\cos^2\theta}\Sigma_{\ind{id}}^{(\ind{H})}\sim
    \Sigma_{\ind{id}}^{(\ind{H})}, \quad e^{2r}+\ind{const}\sim e^{2r},
\end{align}
we may write the effective excess noise covariance matrix
\begin{align}
    \label{eq:sigma_N_eff}
    \Sigma_N^{(\ind{H})\, \ind{eff}}=R(\phi)\,\ind{diag}(\sigma_N,\sigma_N)\,{R}^{\ind{T}}(\phi),
\end{align}
so that enviroment is decribed by vacuum, i.e.
\begin{align}
    \label{eq:sigma-anc-h}
    \Sigma_{\ind{aux}}^{(\ind{H})}=\mathbbm{1}.
\end{align}

Solving for $\theta_{\ind{H}}$ yields
\begin{align}
    \label{eq:cos-theta-h}
    &\cos\theta_{\ind{H}}=\sqrt{\frac{1}{\sigma_x}},\quad
    \sin\theta_{\ind{H}}=\sqrt{\frac{\sigma_N}{\sigma_x}}.
\end{align}

For double homodyne detection, first, we will separate the covariance matrices of sharp measurement and excess noise. From Eq.~\eqref{eq:Sigma_m-DH} and~\eqref{eq:Sigma-id-DH}, we have
\begin{align}
  &\label{eq:Sigma-DH-m-dif}
    \Sigma_{m}^{(\ind{DH})}=\Sigma_{\ind{id}}^{(\ind{DH})}+\Sigma_{N}^{(\ind{DH})}
    ,
\end{align}
where excess noise covariance matrix is given by
\begin{align}
  &
    \label{eq:Sigma-N-DH}
    \Sigma_{N}^{(\ind{DH})}(r)=
    \Sigma_{m}^{(\ind{DH})}-\Sigma_{\ind{id}}^{(\ind{DH})}= R(\phi)
\begin{pmatrix}
      \sigma_N^{(1)}(r) &0\\0&\sigma_N^{(2)}(r)
    \end{pmatrix}
    \tcnj{R}(\phi)\ge 0,\\
  \label{eq:sigma_N-sq-2}
  &
    \sigma_N^{(1)}(r)=\delta_1-e^{2r},
    \quad
    \sigma_N^{(2)}(r)=\delta_2-e^{-2r}.
\end{align}
Note that the elements of $\Sigma_{N}^{(\ind{DH})}$ are dependent on squeezing parameter that lies in the interval~\eqref{eq:r1-r2}.

Now, we note that, since $\Sigma_m^{(\ind{DH})}$ is anisotropic, as generally quadrature variances $\delta_{1,2}$ are not equal, $\delta_1\neq\delta_2$, ancilla state must be anisotropic as well. Without loss of generality, we will consider the ancilla to be in pure state with $\det\Sigma_{\ind{aux}}^{(\ind{DH})}=1$,
\begin{align}
    \Sigma_{\ind{aux}}^{(\ind{DH})}=R(\phi)\,\ind{diag}\left(a,\frac1a\right)\,{R}^{\ind{T}}(\phi),\quad a>0.
\end{align}

From Eq.~\eqref{eq:psi_ast}, we arrive at the expression for $\cos^2\theta_{\ind{DH}}$ in the form
\begin{align}
    \label{eq:cos-theta-dh}
    &\cos^2\theta_{\ind{DH}}(r)={\frac{e^{2r}+a(r)}{\delta_1+a(r)}},\\
    \label{eq:a_def}
    &a(r)=\frac{1}{2}\left[-b+ \sqrt{b^2-4c}\right],\quad
    b=\frac{-\delta_1e^{-2r}+\delta_2e^{2r}}{\sigma_N^{(2)}},\quad c=\frac{-\sigma_N^{(1)}}{\sigma_N^{(2)}}.
\end{align}

It's worth noting that beyond the $r$ interval ancilla becomes strictly unphysical, i.e. $a\in(0,+\infty) \Longleftrightarrow r\in(r_2,r_1)$. This is due to $\sigma_N^{(i)}$ becoming negative, so that the argument of the square root in Eq.~\eqref{eq:a_def} becomes negative as well.

We will also consider classical mixing (additive noise)  channel~\cite{Serafini:bk:2023} used in Ref.~\cite{naumchik2025asymmetryeffectshomodyneheterodyne}:
\begin{align}
  \label{eq:Phi_N}
  &
    \Phi_N:
    \hat{\Pi}_{\ind{id}}\mapsto
    \hat{\Pi}_G= \Phi_N(\hat{\Pi}_{\ind{id}})=
    \frac{2}{\pi\sqrt{\det\Sigma_N}}
    \int_{\mathbb{C}}\dd^2\beta
    \exp\{-2 \vc{s}^T\Sigma_N^{-1}\vc{s}\}
    \hat{D}(\beta)\hat{\Pi}_{\ind{id}}\hat{D}(-\beta),\\
  &\tcnj{\vc{s}}=(\Re(\beta), \Im(\beta))=(\beta_1,\beta_2),\quad
\dd^2\beta=\dd \beta_1\dd \beta_2.
\end{align}
This is a self-dual channel,
$\cnj{\Phi}_N=\Phi_N$,
that transforms the POVM of the ideal projective Gaussian measurements,
$\hat{\Pi}_{\ind{id}}$,
into the POVM of the noisy measurements,
$\hat{\Pi}_{G}$, 
by changing the covariance matrix $\Sigma_{\ind{id}}$
as follows
\begin{align}
  \label{eq:Sigma_N}
  \Phi_N:
  \Sigma_{\ind{id}}\mapsto
  \Sigma_G=\Sigma_{\ind{id}}+\Sigma_N.
\end{align}
\section{GMCS protocol}
In the GMCS protocol, Alice sends coherent states $\ket{\alpha=\alpha_1+i\alpha_2}$ drawn from the Gaussian prior
\begin{align}
  p_A(\alpha)
  =\frac{1}{\pi V_{\ind{A}}}\exp\!\left[-\frac{|\alpha|^2}{2V_{\ind{A}}}\right].
\end{align}
The states pass the composite Gaussian channel $\mathcal{N}_{A\to B}=\Phi_{\xi_{\ind{ch}}}\circ\mathcal{E}_{T_{\ind{ch}}}$, where $\mathcal{E}_{T_{\ind{ch}}}$ is pure loss channel with transmittance $T_{\ind{ch}}$ and $\Phi_{\xi}$ adds excess noise with covariance $\Sigma_{\xi_{\ind{ch}}}=\xi_{\ind{ch}}\mathbbm{1}$. Then, the receiver uses either homodyne or double homodyne detection to measure the states. Mutual information between legitimate parties in each case is then given by the expressions (see Ref.~\cite{naumchik2025asymmetryeffectshomodyneheterodyne} for derivation):
\begin{align}
  \label{eq:I_H_res}
  &I_{\ind{AB}}^{(\ind{H})}=\frac{1}{2}\log\left[1+\frac{4T_{\ind{ch}}V_{\ind{A}}}{\sigma_x+\xi_{\ind{ch}}}\right],\\
  \label{eq:I_DH_res}
  &I_{\ind{AB}}^{(\ind{DH})}=\frac{1}{2}\sum_i\log\left[1+\frac{4T_{\ind{ch}}V_{\ind{A}}}
    {2 \sigma_i+\xi_{\ind{ch}}}\right],
\end{align}
where $\sigma_x$ and $\sigma_i$ given by Eqs.~\eqref{eq:sigma-x} and~\eqref{eq:sigm-x-i}, respectively.

The covariance matrix of TMSV state with the Bob's mode
transmitted through the noisy channel
(see Ref.~\cite{Laudenbach:aqt:2018})
\begin{align}
  \label{eq:TMSVS-to-AB}
  \mathcal{I}_A&\otimes \Phi_{\xi_{\ind{ch}}}\circ\mathcal{E}_{T_{\ind{ch}}}:
\quad
    \Sigma_{\ind{TMSV}}
  =
  \begin{pmatrix}
        V\mathbbm{1}&\sqrt{V^2-1}\sigma_z\\
        \sqrt{V^2-1}\sigma_z&V\mathbbm{1}
    \end{pmatrix}
  \mapsto
    \Sigma_{\ind{AB}}
    =
    \begin{pmatrix}
        V\mathbbm{1}&c\sigma_z\\
        c\sigma_z&V_{\ind{B}}\mathbbm{1}
    \end{pmatrix},
\end{align}
where
$\mathcal{I}_A$ is the identity channel acting on the subsystem $A$
and the parameters of the covariance matrix
$\Sigma_{\ind{AB}}$ are given by
\begin{align}
  \label{eq:V-c}
  &
    V=1+4V_{\ind{A}},
    \quad c=\sqrt{T_{\ind{ch}}(V^2-1)},
  \\
  &
  \label{eq:V_B}
    V_{\ind{B}}=T_{\ind{ch}}(V-1)+1+\xi_{\ind{ch}}.
\end{align}

In untrusted noise scenario, where excess noise caused by asymmetry effects is assumed to be in control of Eve, channel that models the excess noise can be modeled in two equivalent ways, per Eq.~\eqref{eq:E_gamma} and Eq.~\eqref{eq:Phi_N}.

We consider local action of either attenuator channel $\mathcal{E}_N^{(\gamma)}$~\eqref{eq:E_gamma} or additive noise channel $\Phi_N^{(\gamma)}$~\eqref{eq:Phi_N}, on second mode of TMSVS shared by Alice and Bob~\eqref{eq:TMSVS-to-AB}. In both cases, mutual information is given by $I_{\ind{AB}}^{(\gamma)}$~\eqref{eq:I_H_res},~\eqref{eq:I_DH_res}, as it is classical.

It follows
\begin{align}
    \label{eq:psi_action}
    &\mathcal{I}_A\otimes \mathcal{E}_N^{(\gamma)}:
    \Sigma_{\ind{AB}}\mapsto
     \Sigma_{\ind{AB}}^{(\gamma;\, \mathcal{E}_N^{(\gamma)})}
     =\begin{pmatrix}
        V\mathbbm{1}&c\cos\theta_{\gamma}\sigma_z\\
     c\cos\theta_{\gamma}\sigma_z&\cos^2\theta_{\gamma}
     \left(V_{\ind{B}}+\tan^2\theta_{\gamma}\Sigma_{\ind{aux}}^{(\gamma)}\right)
    \end{pmatrix},
\end{align}
\begin{align}
  &
  \label{eq:sigma_m_cov}
\mathcal{I}_A\otimes \Phi_{N}^{(\gamma)}:
  \Sigma_{\ind{AB}}\mapsto
  \Sigma_{\ind{AB}}^{(\gamma;\, \Phi_N^{(\gamma)})}
  =\begin{pmatrix}
    V\mathbbm{1}&c\sigma_z\\
    c\sigma_z&V_{\ind{B}}\mathbbm{1}+\Sigma_N^{(\gamma)}
\end{pmatrix},\quad \gamma\in\{\ind{H},\ind{DH}\}.
\end{align}

When the channel $C^{(\gamma)}\in\{\Phi_N^{(\gamma)}, \mathcal{E}_N^{(\gamma)}\}$ is under control of the eavesdropper, 
Eve holds a purification of the state
$\hat{\rho}_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}=\mathcal{I}_A\otimes C^{(\gamma)}(\hat{\rho}_{\ind{AB}})$ and the measurements performed by the receiver
are assumed to be noiseless.
In this scenario, 
the difference between the joint and conditional entropies
\begin{align}
  \label{eq:chi}
  &
  \chi_{\ind{EB}} ^{(\gamma;\, C^{(\gamma)})}
    =
    S_{\ind{E}}-S_{\ind{E}|\ind{B}}=S_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}-S_{\ind{A}|\ind{B}}^{(\gamma;\, C^{(\gamma)})}, \quad C^{(\gamma)}\in\{\Phi_N^{(\gamma)}, \mathcal{E}_N^{(\gamma)}\}.
\end{align}
gives the Holevo information.
The joint entropy
\begin{align}
  &
  \label{eq:S_AB-gamma}
  S_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}=g(\nu_1^{(\gamma;\, C^{(\gamma)})})+g(\nu_2^{(\gamma;\, C^{(\gamma)})}),
  \\
  &\notag
    g(\nu)=\frac{\nu+1}{2}\log \frac{\nu+1}{2}-
    \frac{\nu-1}{2}\log\frac{\nu-1}{2}
\end{align}
then can be expressed in terms of the symplectic eigenvalues of the covariance matrix~\eqref{eq:sigma_m_cov} given by
\begin{align}
  &
    \label{eq:nu-12-gamma}
\nu_{1,2}^{(\gamma;\, C^{(\gamma)})}
    =\sqrt{
\Delta_{\gamma;\, C^{(\gamma)}}/2\pm \sqrt{\Delta_{\gamma;\, C^{(\gamma)}}^2/4-D_{\gamma;\, C^{(\gamma)}}}
    },
    \quad
  D_{\gamma;\, C^{(\gamma)}}=\det \Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})},  
    \\
  &
  \label{eq:D-gamma}
  \Delta_{\gamma;\, C^{(\gamma)}}=\det([\Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}]_{11})+\det([\Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}]_{22}) +
  2\det([\Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}]_{12}),
\end{align}
where $[\Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}]_{ii}$ ($[\Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}]_{ij}$) are diagonal (off-diagonal) blocks of $\Sigma_{\ind{AB}}^{(\gamma;\, C^{(\gamma)})}$.

Conditional entropies $S_{\ind{E|B}}=S_{\ind{A|B}}^{(\gamma)}$ are calculated from respective conditional covariance matrices,
\begin{align}
  &
    \label{eq:S_A1B-gamma}
    S_{\ind{A|B}}^{(\gamma)}=g(\nu_3^{(\gamma)}),\quad
    \nu_3^{(\ind{\gamma})}=\sqrt{\det\Sigma_{\ind{A|B}}^{(\gamma)}},\\
&  \label{eq:serf-a|b}
  \Sigma_{\ind{A|B}}^{(\gamma)}=
  V\mathbbm{1}-c^2\sigma_z(V_{\ind{B}}\mathbbm{1}+\Sigma^{(\gamma)}_m)^{-1}\sigma_z.
\end{align}
Direct computations show that these conditional matrices and therefore their symplectic eigenvalues are independent on channel model. Symplectic eigenvalues are given by:
\begin{align}
  \label{eq:nu3-H}
    &\nu_3^{(\ind{H})}=\sqrt{V\left(V-\frac{c^2}{V_{\ind{B}}+\sigma_N}\right)},\\
    \label{eq:nu3-DH}
  &
    \nu_3^{(\ind{DH})}=V
    \sqrt{\frac{\left(V_{\ind{B}}  + \delta_1 - \frac{c^{2}}V\right) \left( V_{\ind{B}}  + \delta_2
    - \frac{c^{2}}V\right)}{
    \left( V_{\ind{B}} + \delta_1\right) \left(V_{\ind{B}} + \delta_2\right)}},
\end{align}

From mutual information and Holevo information evaluation of the asymptotic secret fraction is done per Devetak-Winter formula~\cite{Devetak:prsa:2005}
\begin{align}
\label{eq:r}
K^{(\gamma;\, C^{(\gamma)})}=\beta I_{AB}^{(\gamma;\, C^{(\gamma)})}-\chi_{EB}^{(\gamma;\, C^{(\gamma)})},
\quad \gamma\in\{\ind{H},\ind{DH}\}, C^{(\gamma)}\in\{\Phi_N^{(\gamma)}, \mathcal{E}_N^{(\gamma)}\}.
\end{align}

As symplectic eigenvalues for additive noise channel are calculated per formulas Eq.~\eqref{eq:nu-12-gamma} and do not allow for simplification, in the following sections we focus only on analytic expressions for symplectic eigenvalues for attenuator channel model.

\section{Homodyne detection}
For attenuator channel, from Eq.~\eqref{eq:psi_action} and~\eqref{eq:cos-theta-h} we have
\begin{align}
    \label{eq:psi-h-action}
    &\Sigma_{\ind{AB}}^{(\ind{H};\, \mathcal{E}_N^{(\ind{H})})}=
    \begin{pmatrix}
        V\mathbbm{1}&\frac{c}{\sqrt{\sigma_x}}\sigma_z\\
        \frac{c}{\sqrt{\sigma_x}}\sigma_z&\frac{1}{{\sigma_x}}\left(V_{\ind{B}}+\sigma_N\right)\mathbbm{1}
    \end{pmatrix}\equiv
    \begin{pmatrix}
        V\mathbbm{1}&\tilde{c}\sigma_z\\
        \tilde{c}\sigma_z&\tilde{V}_{\ind{B}}\mathbbm{1}
    \end{pmatrix},\\
    &\tilde{V}_{\ind{B}}\equiv
    {T}(V-1)+1+{\xi}
    ,\\
    &\tilde{c}\equiv\sqrt{{T}(V^2-1)},
\end{align}
where we defined scaled Bob's variance and scaled correlations between Alice and Bob $\tilde{V}_{\ind{B}}$ and $\tilde{c}$, using total channel transmission ${T}$ that consists of channel transmission $T_{\ind{ch}}$ and detection losses $\frac{1}{\sigma_x}$, as well as scaled noise variance $\xi$:
\begin{align}
    \label{eq:T_tot_hom}
    &T=T_{\ind{ch}}\cdot \frac{1}{\sigma_x},\\
    \label{eq:xi_tot_hom}
    &\xi\equiv\xi_{\ind{ch}}\cdot\frac{1}{\sigma_x}.
\end{align}

It follows that for homodyne detection the effect of measurement asymmetry is equivalent to scaling both channel transmission~\eqref{eq:T_tot_hom} and channel noise~\eqref{eq:xi_tot_hom} by a factor of $\sigma_x^{-1}$.

Symplectic eigenvalues of $\Sigma_{\ind{AB}}^{(\ind{H};\, \mathcal{E}_N)^{(\ind{H})}}$, $\nu_{1,2}^{(\ind{H};\,\mathcal{E}_N^{(\ind{H})})}$, can be computed using formulas~\cite{Weed:rmp:2012}:
\begin{align}
    \label{eq:lambda12-h}
    \nu_{1,2}^{(\ind{H};\, \mathcal{E}_N^{(\ind{H})})}=\frac{z\pm[\tilde{V}_{\ind{B}}-V]}2,\quad z=\sqrt{\left(V+\tilde{V}_{\ind{B}}\right)^2-4{\tilde{c}^2}},
\end{align}
and then substitued in Eq.~\eqref{eq:S_AB-gamma} to calculate joint entropy $S_{\ind{AB}}^{(\ind{H};\,\mathcal{E}_N^{(\ind{H})})}$.

Fig.~\ref{fig:h-comparison-CM} illustrates the effects of homodyne measurement asymmetry on Holevo information~\eqref{eq:chi} and asymptotic secret fraction~\eqref{eq:r} versus beam splitter transmission $C^2$, as well as secret fraction versus transmission length, for both additive noise~\eqref{eq:sigma_m_cov} and attenuator~\eqref{eq:psi_action} channels.
While channel noise is reduced in this setup~\eqref{eq:xi_tot_hom}, so is channel transmission~\eqref{eq:T_tot_hom}, and, since transmission is scaled by modulation variance $V\gg1$ in every expression, as well as reduced mutual information with greater asymmetry per Eq.~\eqref{eq:I_H_res}, attenuator channel has negative effect. 
As shown in Fig.~\ref{subfig:h-cf-a} and~\ref{subfig:h-cf-b}, the attenuator channel results in higher Holevo information, and thus lower secret fraction, at high transmission for fixed asymmetry parameters. However, Fig.~\ref{subfig:h-cf-c} reveals that the attenuator channel achieves significantly greater reach (maximum transmission length at which a positive key rate is obtained) than the additive noise channel.

For attenuator, from Eq.~\eqref{eq:psi_action} and~\eqref{eq:cos-theta-dh} we have
\begin{align}
    &\Sigma_{\ind{AB}}^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}=
    \notag\\
    &\tiny
    \begin{pmatrix}
        V\mathbbm{1}&{c}\cos\theta^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}\sigma_z\\
        c\cos\theta^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}\sigma_z&
        \begin{matrix}
            \cos^2\theta^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}\left({V_{\ind{B}}}+\delta_1-\frac{e^{2r}}{\cos^2\theta^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}}\right)&0\\
            0&\cos^2\theta^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}\left({V_{\ind{B}}}+\delta_2-\frac{e^{-2r}}{\cos^2\theta^{(\ind{DH};\, \mathcal{E}_N^{(\ind{DH})})}}\right)
        \end{matrix}
    \end{pmatrix}.
    \end{align}

Symplectic eigenvalues of this matrix can be calculated using invariants similarly to Eq.~\eqref{eq:nu-12-gamma}. It is obvious that these symplectic eigenvalues, and thereby joint entropy, is explicitly depended on squeezing parameter $r$~\eqref{eq:r1-r2}, like in the case of additive noise channel. 

Fig.~\ref{fig:Entropy} illustrates dependence of Holevo information on the squeezing parameter, for both additive noise and attenuator channels. Referring to Fig.~\ref{fig:Entropy}, in the limiting case of ideal double homodyne with $\delta_1=\delta_2=1$, only one value of $r=0$ is allowed (see Ref.~\cite{naumchik2025asymmetryeffectshomodyneheterodyne}), with equal values of Holevo information for both channels (black dot on plot).

When $\delta_1=\delta_2>1$, dependence of Holevo information on squeezing parameter is symmetrical around $r=0$ for both channels. For additive noise channel, $r=0$ is the point of maximal Holevo information, while for attenuator channel $r=0$ corresponds to minimal Holevo information. Moreover, while Holevo information is unidefined at $r=r_{2,1}$ (see Eq.~\eqref{eq:cos-theta-dh}), the values tend to the same values of joint entropy in the case of additive noise channel, as signified by color-matched white-filled dots in Fig.~\ref{fig:Entropy}.

In asymmetrical case where $\delta_1\neq\delta_2$, Holevo information for both channel models become asymmetrical relative to their extrema, resulting in unequal joint entropies at $r=r_{2,1}$. For additive noise channel, Holevo information has a distintive maxima, while for attenuator channel Holevo information has a distintive minima.
